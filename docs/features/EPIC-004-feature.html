buildAdmissionCycles(d) { const KEY = 'geta_admission_cycles'; const DEFAULT_PROGRAMMES = ['BSc AI', 'MBA', 'CSE']; const YEARS = ['2024-25', '2025-26', '2026-27', '2027-28']; const seed = [ { id: 'ac1', name: 'MBA Admission 2026-27', programme: 'MBA', academicYear: '2026-27', status: 'Active', candidates: 142 }, { id: 'ac2', name: 'CSE Admission 2026-27', programme: 'CSE', academicYear: '2026-27', status: 'Draft', candidates: 0 }, { id: 'ac3', name: 'MBA Admission 2025-26', programme: 'MBA', academicYear: '2025-26', status: 'Closed', candidates: 118 } ]; let cycles = JSON.parse(localStorage.getItem(KEY) || 'null'); if (!cycles) { cycles = seed; localStorage.setItem(KEY, JSON.stringify(cycles)); } // ponytail: read live programmes from d.programmes if available, else hardcode const liveProgrammes = (d && d.programmes && d.programmes.length) ? d.programmes.map(p => p.name) : DEFAULT_PROGRAMMES; const s = this.state; const filterProg = s.acFilterProg || ''; const filterYear = s.acFilterYear || ''; const filterStatus = s.acFilterStatus || ''; const showCreate = !!s.acShowCreate; const newName = s.acNewName || ''; const newProg = s.acNewProg || ''; const newYear = s.acNewYear || ''; const newStatus = s.acNewStatus || 'Draft'; const save = (updated) => { localStorage.setItem(KEY, JSON.stringify(updated)); this.setState({ acRev: (s.acRev || 0) + 1 }); }; const filtered = cycles.filter(c => (!filterProg || c.programme === filterProg) && (!filterYear || c.academicYear === filterYear) && (!filterStatus || c.status === filterStatus) ); const cycleRows = filtered.map(c => ({ id: c.id, name: c.name, programme: c.programme, academicYear: c.academicYear, candidates: c.candidates, isActive: c.status === 'Active', isDraft: c.status === 'Draft', isClosed: c.status === 'Closed', canActivate: c.status === 'Draft', canClose: c.status === 'Active', onActivate: () => { const all = JSON.parse(localStorage.getItem(KEY)); const row = all.find(x => x.id === c.id); if (row) { row.status = 'Active'; save(all); this.showToast('Cycle activated.'); } }, onClose: () => { const all = JSON.parse(localStorage.getItem(KEY)); const row = all.find(x => x.id === c.id); if (row) { row.status = 'Closed'; save(all); this.showToast('Cycle closed.'); } }, onDelete: () => { const all = JSON.parse(localStorage.getItem(KEY)).filter(x => x.id !== c.id); save(all); this.showToast('Cycle deleted.'); } })); const programmeOptions = liveProgrammes.map(p => ({ value: p, label: p, selected: filterProg === p, selectedForForm: newProg === p })); const yearOptions = YEARS.map(y => ({ value: y, label: y, selected: filterYear === y, selectedForForm: newYear === y })); return { showCreateForm: showCreate, noCycles: cycleRows.length === 0, cycleRows, programmeOptions, yearOptions, newCycleName: newName, onToggleCreateForm: () => this.setState({ acShowCreate: !showCreate, acNewName: '', acNewProg: '', acNewYear: '', acNewStatus: 'Draft' }), onFilterProgramme: (e) => this.setState({ acFilterProg: e.target.value }), onFilterYear: (e) => this.setState({ acFilterYear: e.target.value }), onFilterStatus: (e) => this.setState({ acFilterStatus: e.target.value }), onNewCycleName: (e) => this.setState({ acNewName: e.target.value }), onNewCycleProgramme: (e) => this.setState({ acNewProg: e.target.value }), onNewCycleYear: (e) => this.setState({ acNewYear: e.target.value }), onNewCycleStatus: (e) => this.setState({ acNewStatus: e.target.value }), onCreateCycle: () => { if (!newName.trim() || !newProg || !newYear) { this.showToast('Please fill Cycle Name, Programme, and Academic Year.'); return; } const all = JSON.parse(localStorage.getItem(KEY)); all.push({ id: 'ac' + Date.now(), name: newName.trim(), programme: newProg, academicYear: newYear, status: newStatus, candidates: 0 }); save(all); this.setState({ acShowCreate: false, acNewName: '', acNewProg: '', acNewYear: '', acNewStatus: 'Draft' }); this.showToast('Admission cycle created.'); } }; }
